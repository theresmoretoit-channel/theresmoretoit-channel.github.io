# Neden ışıktan hızlı yolculuk edemiyoruz?

<!-- S01 -->

gönderebildiğinizi düşünün.

<!-- S02 -->

<!-- S05 -->

<!-- S16 -->

<!-- S17 -->

<!-- S18 -->

trenin ortasında oturuyor.
Ama o, öndeki şimşeğe doğru

<!-- S19 -->

<!-- S20 -->

<!-- S28 -->

<!-- S30 -->

<!-- S32 -->

<!-- S35 -->

<!-- S41b -->
