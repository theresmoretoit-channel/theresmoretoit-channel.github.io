# Co kdyby se jedenácté září nikdy nestalo?

<!-- S10 -->

<!-- S11 -->

<!-- S12 -->

<!-- S13b -->
A ten rozdíl

<!-- S15 -->

<!-- S16 -->

<!-- S17 -->

<!-- S21 -->

Možná žádná okupace.

<!-- S27 -->

<!-- S28 -->
A ta obrovská bezpečnostní mašinerie,

<!-- S28b -->

<!-- S29 -->

<!-- S30 -->
